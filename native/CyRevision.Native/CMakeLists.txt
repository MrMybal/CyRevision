cmake_minimum_required(VERSION 3.20)
project(CyRevisionNative VERSION 2.0.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The native V2 renderer currently requires Windows and Direct3D 11.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CYUI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/CyUI" CACHE PATH "CyUI source directory")
set(JSON_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/../../.tools/native-json/json.hpp" CACHE FILEPATH "Pinned JSON single header")
if(NOT EXISTS "${CYUI_ROOT}/include/cyui/CyUI.h")
    message(FATAL_ERROR "Set CYUI_ROOT to the CyUI source directory.")
endif()
if(NOT EXISTS "${JSON_HEADER}")
    message(FATAL_ERROR "Run scripts/build-native-v2.ps1 to prepare native dependencies.")
endif()
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
if(MSVC)
    add_compile_options(/experimental:deterministic "/pathmap:${CMAKE_CURRENT_SOURCE_DIR}/../../=/_/" "/pathmap:${CYUI_ROOT}=/_/native/vendor/CyUI")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../Directory.Build.props" VERSION_PROPS)
string(REGEX MATCH "<VersionPrefix>([0-9]+)\\.([0-9]+)\\.([0-9]+)</VersionPrefix>" VERSION_MATCH "${VERSION_PROPS}")
if(NOT VERSION_MATCH)
    message(FATAL_ERROR "The application version is missing.")
endif()
set(CYREVISION_VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
set(CYREVISION_FILE_VERSION "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},0")
configure_file(NativeVersion.h.in "${CMAKE_CURRENT_BINARY_DIR}/include/NativeVersion.h" @ONLY)
add_subdirectory("${CYUI_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/cyui")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/nlohmann")
configure_file("${JSON_HEADER}" "${CMAKE_CURRENT_BINARY_DIR}/include/nlohmann/json.hpp" COPYONLY)
set(IMGUI_DIR "${CYUI_ROOT}/third_party/imgui")
set(CYREVISION_WORKSPACE_SOURCES Client.cpp Workspace.cpp Panels.cpp DetachedViews.cpp Inspector.cpp StringInput.cpp NativeWidgets.cpp Assets.cpp Shortcuts.cpp GitViews.cpp GitLfsViews.cpp PullRequestViews.cpp CiViews.cpp ToolsViews.cpp OverviewViews.cpp PreferencesView.cpp GitConflictViews.cpp)
add_executable(CyRevision.Native WIN32 Main.cpp NativeTray.cpp ${CYREVISION_WORKSPACE_SOURCES} Branding.rc
    "${IMGUI_DIR}/backends/imgui_impl_win32.cpp"
    "${IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
target_compile_features(CyRevision.Native PRIVATE cxx_std_17)
target_compile_definitions(CyRevision.Native PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(CyRevision.Native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include" "${IMGUI_DIR}/backends" "${IMGUI_DIR}")
target_link_libraries(CyRevision.Native PRIVATE CyUI::CyUI d3d11 dxgi d3dcompiler dwmapi user32 gdi32 shell32 ole32 uuid windowscodecs)
if(MSVC)
    target_compile_options(CyRevision.Native PRIVATE /W4 /utf-8 /EHsc)
    target_link_options(CyRevision.Native PRIVATE /DEBUG:NONE)
endif()
install(TARGETS CyRevision.Native RUNTIME DESTINATION .)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../src/CyRevision.Desktop/Assets/Branding/cyrevision-cy-simple-badge-transparent.png" DESTINATION . RENAME cyrevision-logo.png)
install(FILES "${CYUI_ROOT}/LICENSE" DESTINATION Licenses RENAME CyUI.txt)
install(FILES "${IMGUI_DIR}/LICENSE.txt" DESTINATION Licenses RENAME DearImGui.txt)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../Licenses/nlohmann-json.txt" DESTINATION Licenses)

option(CYREVISION_BUILD_NATIVE_TESTS "Build the headless native UI regression tests" ON)
if(CYREVISION_BUILD_NATIVE_TESTS)
    enable_testing()
    add_executable(CyRevision.Native.UiTests tests/NativeUiTests.cpp ${CYREVISION_WORKSPACE_SOURCES})
    target_compile_features(CyRevision.Native.UiTests PRIVATE cxx_std_17)
    target_compile_definitions(CyRevision.Native.UiTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(CyRevision.Native.UiTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/include")
    target_link_libraries(CyRevision.Native.UiTests PRIVATE CyUI::CyUI shell32 ole32 uuid d3d11 windowscodecs)
    if(MSVC)
        target_compile_options(CyRevision.Native.UiTests PRIVATE /W4 /utf-8 /EHsc)
        target_link_options(CyRevision.Native.UiTests PRIVATE /DEBUG:NONE)
    endif()
    add_test(NAME NativeUiInput COMMAND CyRevision.Native.UiTests)
    add_executable(CyRevision.Native.TrayTests tests/NativeTrayTests.cpp NativeTray.cpp)
    target_compile_features(CyRevision.Native.TrayTests PRIVATE cxx_std_17)
    target_compile_definitions(CyRevision.Native.TrayTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(CyRevision.Native.TrayTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(CyRevision.Native.TrayTests PRIVATE shell32 user32)
    if(MSVC)
        target_compile_options(CyRevision.Native.TrayTests PRIVATE /W4 /utf-8 /EHsc)
        target_link_options(CyRevision.Native.TrayTests PRIVATE /DEBUG:NONE)
    endif()
    add_test(NAME NativeTrayLifecycle COMMAND CyRevision.Native.TrayTests)
endif()
