cmake_minimum_required(VERSION 3.24)
project(CyRevisionCyMcp LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CYMCP_BUILD_HTTP ON CACHE BOOL "" FORCE)
set(CYMCP_BUILD_PLATFORM ON CACHE BOOL "" FORCE)
set(CYMCP_BUILD_MCP ON CACHE BOOL "" FORCE)
set(CYMCP_BUILD_AI ON CACHE BOOL "" FORCE)
set(CYMCP_BUILD_LOCAL OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_CHAT_UI OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_CHAT_APP OFF CACHE BOOL "" FORCE)
if(MSVC)
  add_compile_options(/utf-8 /experimental:deterministic "/pathmap:${CMAKE_CURRENT_SOURCE_DIR}/../../=/_/")
else()
  get_filename_component(CYREVISION_SOURCE_ROOT "../.." ABSOLUTE)
  add_compile_options("-ffile-prefix-map=${CYREVISION_SOURCE_ROOT}=/_/" "-fmacro-prefix-map=${CYREVISION_SOURCE_ROOT}=/_/")
endif()
add_subdirectory(../vendor/CyMCPSDK "${CMAKE_CURRENT_BINARY_DIR}/sdk")
add_library(CyRevision.CyMcpNative SHARED Bridge.cpp)
target_compile_features(CyRevision.CyMcpNative PRIVATE cxx_std_20)
target_link_libraries(CyRevision.CyMcpNative PRIVATE CyMCP::Core CyMCP::Platform CyMCP::Http CyMCP::MCP CyMCP::AI)
if(MSVC)
  target_compile_definitions(CyRevision.CyMcpNative PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_options(CyRevision.CyMcpNative PRIVATE /DEBUG:NONE)
endif()
install(TARGETS CyRevision.CyMcpNative RUNTIME DESTINATION . COMPONENT CyRevisionCyMcp LIBRARY DESTINATION . COMPONENT CyRevisionCyMcp)
install(FILES ../vendor/CyMCPSDK/LICENSE DESTINATION Licenses RENAME CyMCPSDK.txt COMPONENT CyRevisionCyMcp)
install(FILES ../vendor/CyMCPSDK/THIRD_PARTY_NOTICES.md DESTINATION Licenses RENAME CyMCPSDK-Third-Party.md COMPONENT CyRevisionCyMcp)
install(FILES ../vendor/CyMCPSDK/third_party/nlohmann/LICENSE.MIT DESTINATION Licenses RENAME CyMCPSDK-Json.txt COMPONENT CyRevisionCyMcp)
install(FILES ../vendor/CyMCPSDK/third_party/httplib/LICENSE DESTINATION Licenses RENAME CyMCPSDK-Http.txt COMPONENT CyRevisionCyMcp)
install(FILES ../vendor/CyMCPSDK/third_party/picosha2/LICENSE DESTINATION Licenses RENAME CyMCPSDK-Sha.txt COMPONENT CyRevisionCyMcp)
